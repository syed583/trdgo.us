function G { param([Parameter(ValueFromRemainingArguments = $true)]$a) @(& git --no-optional-locks -C $Root @a 2>&1) }
function State {
    $s = Get-CimInstance Win32_Service -Filter ("Name='" + $Svc + "'")
    $h = $null; try { $h = (Invoke-WebRequest -Uri ("http://127.0.0.1:" + $Port + "/health") -UseBasicParsing -TimeoutSec 8).Content } catch { }
    [pscustomobject]@{ Head = ((G rev-parse HEAD) -join '').Trim(); Branch = ((G branch --show-current) -join '').Trim(); Dirty = @(G status --porcelain | Where-Object { $_ }).Count
        Stash = @(G stash list | Where-Object { $_ }).Count; State = $(if ($s) { $s.State } else { 'NOT FOUND' }); Pid = $(if ($s) { $s.ProcessId } else { $null }); Healthy = [bool]($h -and $h -match '"status"\s*:\s*"ok"')
        Started = $(if ($s -and $s.ProcessId) { (Get-Process -Id $s.ProcessId -ErrorAction SilentlyContinue).StartTime } else { $null }) }
}
$bytes = [Convert]::FromBase64String(($b64 -replace '\s', ''))
$sha = [BitConverter]::ToString([Security.Cryptography.SHA256]::Create().ComputeHash($bytes)).Replace('-', '')
if ($sha -ne 'C34E8FAF944C0A8F26AA170730000487FF3069CF3FB6F02293ADF8751A0A3EFA') { Write-Host "HASH MISMATCH ($sha): nothing written." -ForegroundColor Red; return }
Write-Host "[PASS] SHA-256 matches C34E8FAF944C0A8F26AA170730000487FF3069CF3FB6F02293ADF8751A0A3EFA"
$tmp = Join-Path $env:TEMP 'after_close_syntax.ps1'; [IO.File]::WriteAllBytes($tmp, $bytes)
$tk = $null; $er = $null; [void][System.Management.Automation.Language.Parser]::ParseFile($tmp, [ref]$tk, [ref]$er); Remove-Item $tmp -ErrorAction SilentlyContinue

$null = New-Item -ItemType Directory -Force -Path $Dir
$f = Join-Path $Dir 'phaseB_after_close_check.ps1'
if (Test-Path $f) {
    if ((Get-FileHash -LiteralPath $f -Algorithm SHA256).Hash -eq $sha) { Write-Host "already installed and identical: nothing to write" }
    else { $old = $f + '.old-' + (Get-Date -Format 'yyyyMMdd-HHmmss'); Rename-Item -LiteralPath $f -NewName (Split-Path $old -Leaf); Write-Host ("an existing different file was renamed (kept): " + $old) }
}
if (-not (Test-Path $f)) { [IO.File]::WriteAllBytes($f, $bytes) }
$back = (Get-FileHash -LiteralPath $f -Algorithm SHA256).Hash
if ($back -eq $sha) { Write-Host ("[PASS] installed and read back identical: " + $f) } else { Write-Host "[FAIL] installed file does not match" -ForegroundColor Red }
